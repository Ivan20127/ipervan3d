const canvas = document.getElementById("c"); const ctx = canvas.getContext("2d"); let keys = {}; addEventListener("keydown", e => keys[e.key] = true); addEventListener("keyup", e => keys[e.key] = false); // SAFE SOUND (NO FREEZE) let audioCtx; function beep(freq){ if(!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); const o = audioCtx.createOscillator(); const g = audioCtx.createGain(); o.connect(g); g.connect(audioCtx.destination); o.frequency.value = freq; g.gain.value = 0.1; o.start(); g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1); o.stop(audioCtx.currentTime + 0.1); } // PLAYER let player = { x:100, y:300, vx:0, vy:0, w:18, h:40, onGround:false, hp:3, lives:3 }; // LEVEL PLATFORM (ALL ENEMIES CAN REACH THESE) const platforms = [ [0,450,900,50], [150,380,200,10], [400,320,200,10], [650,260,200,10] ]; // YOUR PRODUCTS AS ENEMIES const productNames = [ "Gear Fidget","Dragon","Pig","Otter","Shark", "Knight","Cactus","Air Tank" ]; // ENEMIES let enemies = []; function spawnEnemies(){ enemies = []; for(let i=0;i<6;i++){ enemies.push({ x:200 + i*100, y:420, w:25,h:25, vx:0, speed:1.5, name: productNames[Math.floor(Math.random()*productNames.length)] }); } } spawnEnemies(); // COLLISION function hit(a,b){ return a.x < b.x+b.w && a.x+a.w > b.x && a.y < b.y+b.h && a.y+a.h > b.y; } // JUMP function jump(){ if(player.onGround){ player.vy = -11; player.onGround = false; beep(300); } } // DAMAGE function damage(){ player.hp--; beep(120); if(player.hp <= 0){ player.lives--; player.hp = 3; if(player.lives <= 0){ alert("GAME OVER"); location.reload(); } } } // UPDATE function update(){ // player movement if(keys["ArrowLeft"]) player.vx = -4; else if(keys["ArrowRight"]) player.vx = 4; else player.vx *= 0.8; // gravity player.vy += 0.5; player.x += player.vx; player.y += player.vy; // ground if(player.y > 420){ player.y = 420; player.vy = 0; player.onGround = true; } // platforms (solid) for(let p of platforms){ let plat = {x:p[0],y:p[1],w:p[2],h:p[3]}; if(hit(player,plat) && player.vy > 0){ player.y = plat.y - player.h; player.vy = 0; player.onGround = true; } } // ENEMY AI (FIXED: always reachable + stable) for(let e of enemies){ // CHASE PLAYER (REAL AI) if(player.x < e.x) e.x -= e.speed; else e.x += e.speed; // STAY ON GROUND (IMPORTANT FIX) e.y = 420; // ATTACK RANGE if(Math.abs(player.x - e.x) < 25 && Math.abs(player.y - e.y) < 40){ damage(); } // STOMP KILL if(player.vy > 0 && hit(player,e)){ e.dead = true; player.vy = -8; beep(200); } } // CLEAN ENEMIES enemies = enemies.filter(e => !e.dead); // NEXT WAVE if(enemies.length === 0){ spawnEnemies(); } // bounds player.x = Math.max(0, Math.min(880, player.x)); } // DRAW function draw(){ ctx.clearRect(0,0,900,500); // platforms ctx.fillStyle="#334155"; for(let p of platforms){ ctx.fillRect(p[0],p[1],p[2],p[3]); } // player (stick figure) ctx.strokeStyle="white"; ctx.lineWidth=3; ctx.beginPath(); ctx.arc(player.x,player.y-20,10,0,Math.PI*2); ctx.stroke(); ctx.beginPath(); ctx.moveTo(player.x,player.y-10); ctx.lineTo(player.x,player.y+20); ctx.stroke(); ctx.beginPath(); ctx.moveTo(player.x,player.y+20); ctx.lineTo(player.x-10,player.y+40); ctx.moveTo(player.x,player.y+20); ctx.lineTo(player.x+10,player.y+40); ctx.stroke(); // enemies (PRODUCTS) ctx.fillStyle="#f97316"; for(let e of enemies){ ctx.fillRect(e.x,e.y,25,25); ctx.fillStyle="white"; ctx.font="10px Arial"; ctx.fillText(e.name,e.x-15,e.y-5); ctx.fillStyle="#f97316"; } } // LOOP function loop(){ update(); draw(); requestAnimationFrame(loop); } addEventListener("keydown",e=>{ if(e.key===" ") jump(); }); loop();
